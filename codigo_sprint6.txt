<script setup>
import { ref, computed, watch } from 'vue';
import { authService } from '../services/authService';
import { recipeService } from '../services/recipeService';

const props = defineProps({
  recipe: {
    type: Object,
    required: true
  }
});

defineEmits(['back']);

const newComment = ref('');
const commentError = ref('');
const ratingError = ref('');
const ratingMessage = ref('');
const selectedRating = ref(0);

const ratingSummary = computed(() =>
  recipeService.getRatingSummary(props.recipe.id)
);

const loadUserRating = () => {
  selectedRating.value = recipeService.getUserRating(
    props.recipe.id
  );
};

watch(
  () => props.recipe.id,
  loadUserRating,
  { immediate: true }
);

const submitComment = () => {
  commentError.value = '';

  try {
    recipeService.addComment(
      props.recipe.id,
      newComment.value
    );

    newComment.value = '';
  } catch (error) {
    commentError.value = error.message;
  }
};

const deleteComment = (commentId) => {
  commentError.value = '';

  if (!confirm('¿Deseas eliminar este comentario?')) {
    return;
  }

  try {
    recipeService.deleteComment(
      props.recipe.id,
      commentId
    );
  } catch (error) {
    commentError.value = error.message;
  }
};

const submitRating = (score) => {
  ratingError.value = '';
  ratingMessage.value = '';

  try {
    recipeService.rateRecipe(props.recipe.id, score);
    selectedRating.value = score;
    ratingMessage.value = 'Tu calificación fue guardada.';
  } catch (error) {
    ratingError.value = error.message;
  }
};

const canDeleteComment = (comment) =>
  authService.currentUser.value?.id === comment.userId;

const formatDate = (date) =>
  new Intl.DateTimeFormat('es-MX', {
    dateStyle: 'medium',
    timeStyle: 'short'
  }).format(new Date(date));
</script>

<template>
  <div class="create-container">
    <h2>Publicar Nueva Receta</h2>

      <input v-model="form.title" type="text" placeholder="Título de la receta" required />
      
      <div class="row">
        <select v-model="form.category" required>
          <option value="" disabled>Selecciona Categoría</option>
          <option value="Pasta">Pasta</option>
          <option value="Mexicana">Mexicana</option>
          <option value="Ensaladas">Ensaladas</option>
          <option value="Postres">Postres</option>
        </select>

        <input v-model="form.time" type="number" placeholder="Tiempo (mins)" min="5" required />

        <select v-model="form.difficulty" required>
          <option value="" disabled>Dificultad</option>
          <option value="Fácil">Fácil</option>
          <option value="Media">Media</option>
          <option value="Difícil">Difícil</option>
        </select>
      </div>

      <textarea v-model="form.description" placeholder="Breve descripción" rows="2" required></textarea>

      <div class="dynamic-section">
        <h3>Fotografía de la Receta</h3>
        <input type="file" @change="handleFileUpload" accept="image/png, image/jpeg" />
        <p v-if="imageError" class="error">{{ imageError }}</p>
      </div>

        <h3>Ingredientes</h3>
        <div v-for="(ing, idx) in form.ingredients" :key="idx" class="dynamic-row">
          <input v-model="form.ingredients[idx]" placeholder="Ej. 200g de harina" required />
          <button type="button" @click="removeIngredient(idx)" class="btn-remove">X</button>
        </div>
        <button type="button" @click="addIngredient" class="btn-add">+ Agregar Ingrediente</button>
      </div>

      <!-- Lista dinámica de Pasos -->
      <div class="dynamic-section">
        <h3>Pasos de Preparación</h3>
        <div v-for="(step, idx) in form.steps" :key="idx" class="dynamic-row">
          <input v-model="form.steps[idx]" placeholder="Ej. Mezclar los ingredientes secos" required />
          <button type="button" @click="removeStep(idx)" class="btn-remove">X</button>
        </div>
        <button type="button" @click="addStep" class="btn-add">+ Agregar Paso</button>
      </div>

      <button type="submit" class="btn-submit">Guardar y Publicar Receta</button>
    </form>
  </div>

<script setup>
import { reactive } from 'vue';
import { recipeService } from '../services/recipeService';
import { ref } from 'vue';

const imageError = ref('');

const handleFileUpload = (event) => {
  const file = event.target.files[0];
  if (file) {
    if (file.size > 2 * 1024 * 1024) { // Límite de 2MB
      imageError.value = 'La imagen es muy pesada. Máximo 2MB.';
      return;
    }
    imageError.value = '';
    // Simulamos la conversión a URL local para previsualizar y guardar
    form.image = URL.createObjectURL(file);
  }
};

const emit = defineEmits(['recipe-created']);

const form = reactive({
  title: '',
  category: '',
  time: '',
  difficulty: '',
  description: '',
  ingredients: [''],
  steps: [''],
  image: ''
});

const addIngredient = () => form.ingredients.push('');
const removeIngredient = (idx) => form.ingredients.splice(idx, 1);

const addStep = () => form.steps.push('');
const removeStep = (idx) => form.steps.splice(idx, 1);

const submitRecipe = async () => {
  try {
    await recipeService.create({
      ...form,
      ingredients: [...form.ingredients],
      steps: [...form.steps]
    });

    alert('Receta publicada exitosamente.');
    emit('recipe-created');
  } catch (error) {
    alert(error.message);
  }
};
</script>

      <textarea v-model="form.description" placeholder="Descripción" rows="3" required></textarea>
      
      <div class="actions">
        <button type="submit" class="btn-save">Guardar Cambios</button>
        <button type="button" @click="handleDelete" class="btn-delete">Eliminar Receta</button>
      </div>
    </form>
  </div>

<script setup>
import { reactive } from 'vue';
import { useRecipeStore } from '../stores/recipeStore';

const props = defineProps({ recipe: Object });
const emit = defineEmits(['close']);
const store = useRecipeStore();

const form = reactive({ ...props.recipe });

const handleUpdate = () => {
  store.updateRecipe(form);
  alert('¡Receta actualizada con éxito!');
  emit('close');
};

const handleDelete = () => {
  if (confirm('¿Estás seguro de eliminar esta receta?')) {
    store.deleteRecipe(form.id);
    alert('Receta eliminada.');
    emit('close');
  }
};
</script>

  query: '',

  maxTime: 90

  filters.value = newFilters;
};

const filteredRecipes = computed(() => {
  return recipes.value.filter((r) => {
    const matchesQuery =
      !filters.value.query ||
      r.title.toLowerCase().includes(filters.value.query.toLowerCase()) ||
      r.ingredients.some((i) => i.toLowerCase().includes(filters.value.query.toLowerCase()));

    const matchesCategory = !filters.value.category || r.category === filters.value.category;
    const matchesDifficulty = !filters.value.difficulty || r.difficulty === filters.value.difficulty;
    const matchesTime = r.time <= filters.value.maxTime;

    return matchesQuery && matchesCategory && matchesDifficulty && matchesTime;
  });
});

import { ref, computed } from 'vue';

const STORAGE_KEY = 'chefcompartido_session';

const savedSession = localStorage.getItem(STORAGE_KEY);

const currentUser = ref(
  savedSession ? JSON.parse(savedSession).user : null
);

const token = ref(
  savedSession ? JSON.parse(savedSession).token : null
);

const saveSession = () => {
  if (!currentUser.value || !token.value) {
    localStorage.removeItem(STORAGE_KEY);
    return;
  }

  localStorage.setItem(
    STORAGE_KEY,
    JSON.stringify({
      user: currentUser.value,
      token: token.value
    })
  );
};

export const authService = {
  currentUser,
  token,

  isAuthenticated: computed(() => Boolean(currentUser.value && token.value)),

  login(email, password) {
    if (!email || !password) {
      return Promise.reject(
        new Error('El correo y la contraseña son obligatorios.')
      );
    }

    currentUser.value = {
      id: email.toLowerCase(),
      name: email.split('@')[0],
      email
    };

    token.value = `demo-token-${Date.now()}`;
    saveSession();

    return Promise.resolve({
      user: currentUser.value,
      token: token.value
    });
  },

  register(userData) {
    if (!userData.name || !userData.email || !userData.password) {
      return Promise.reject(
        new Error('Todos los datos son obligatorios.')
      );
    }

    return Promise.resolve({
      success: true,
      message: 'Usuario registrado con éxito'
    });
  },

  logout() {
    currentUser.value = null;
    token.value = null;
    localStorage.removeItem(STORAGE_KEY);
  },

  requireAuth() {
    if (!this.isAuthenticated.value) {
      throw new Error('Debes iniciar sesión para realizar esta acción.');
    }

    return currentUser.value;
  }
};

const normalizeRecipe = (recipe) => ({
  ...recipe,
  comments: Array.isArray(recipe.comments) ? recipe.comments : [],

const loadRecipes = () => {

const recipes = ref(loadRecipes());

const findRecipe = (recipeId) => {
  const recipe = recipes.value.find(
    (item) => item.id === Number(recipeId)
  );

  if (!recipe) {

export const recipeService = {
  recipes,

  getAll() {
    return recipes.value;
  },

  getById(id) {
    return recipes.value.find(
      (recipe) => recipe.id === Number(id)
    );
  },

  create(newRecipe) {
    const user = authService.requireAuth();

    const created = normalizeRecipe({
      ...newRecipe,
      id: Date.now(),
      time: Number(newRecipe.time),
      authorId: user.id,
      author: user.name
    });

    recipes.value.unshift(created);
    saveRecipes();

    return Promise.resolve(created);
  },

  addComment(recipeId, text) {
    const user = authService.requireAuth();
    const cleanText = text.trim();

    if (cleanText.length < 3) {
      throw new Error(
        'El comentario debe contener al menos 3 caracteres.'
      );
    }

    if (cleanText.length > 500) {
      throw new Error(
        'El comentario no puede superar los 500 caracteres.'
      );
    }

    const recipe = findRecipe(recipeId);

    const comment = {
      id: Date.now(),
      userId: user.id,
      userName: user.name,
      text: cleanText,
      createdAt: new Date().toISOString()
    };

    recipe.comments.unshift(comment);
    saveRecipes();

    return comment;
  },

  deleteComment(recipeId, commentId) {
    const user = authService.requireAuth();
    const recipe = findRecipe(recipeId);

    const comment = recipe.comments.find(
      (item) => item.id === Number(commentId)
    );

    if (!comment) {
      throw new Error('No se encontró el comentario.');
    }

    if (comment.userId !== user.id) {
      throw new Error(
        'Solo puedes eliminar tus propios comentarios.'
      );
    }

    recipe.comments = recipe.comments.filter(
      (item) => item.id !== Number(commentId)
    );

    saveRecipes();
  },

  rateRecipe(recipeId, score) {
    const user = authService.requireAuth();
    const numericScore = Number(score);

    if (
      !Number.isInteger(numericScore) ||
      numericScore < 1 ||
      numericScore > 5
    ) {
      throw new Error(
        'La calificación debe estar entre 1 y 5 estrellas.'
      );
    }

    const recipe = findRecipe(recipeId);

    const existingRating = recipe.ratings.find(
      (rating) => rating.userId === user.id
    );

    if (existingRating) {
      existingRating.score = numericScore;
      existingRating.updatedAt = new Date().toISOString();
    } else {
      recipe.ratings.push({
        userId: user.id,
        userName: user.name,
        score: numericScore,
        createdAt: new Date().toISOString()
      });
    }

    saveRecipes();

    return this.getRatingSummary(recipeId);
  },

  getUserRating(recipeId) {
    if (!authService.currentUser.value) {
      return 0;
    }

    const recipe = findRecipe(recipeId);

    return (
      recipe.ratings.find(
        (rating) =>
          rating.userId === authService.currentUser.value.id
      )?.score || 0
    );
  },

  getRatingSummary(recipeId) {
    const recipe = findRecipe(recipeId);
    const total = recipe.ratings.length;

    if (total === 0) {
      return {
        average: 0,
        total: 0
      };
    }

    const sum = recipe.ratings.reduce(
      (accumulator, rating) =>
        accumulator + Number(rating.score),
      0
    );

    return {
      average: Number((sum / total).toFixed(1)),
      total
    };
  }
};
